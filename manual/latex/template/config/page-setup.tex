% Controle de fundo das páginas internas: papel timbrado institucional para o miolo padrão
\newif\ifshowcontentbg
\showcontentbgtrue

% Inclusão do papel timbrado oficial via fancyhdr (sem dependência de pacotes externos)
\fancypagestyle{editorialcontent}{%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
  \fancyhead[L]{%
    \ifshowcontentbg
      \begin{tikzpicture}[remember picture, overlay]
        \node[inner sep=0pt, outer sep=0pt] at (current page.center) {%
          \includegraphics[width=\paperwidth, height=\paperheight]{capas/papel-timbrado-miolo.pdf}%
        };
      \end{tikzpicture}%
    \fi
  }%
  % Numeração de página discreta e elegante no canto inferior direito
  \fancyfoot[R]{\fontsize{10.5pt}{13pt}\fontSemiBold\textcolor{BrandGrayMed}{\thepage}}%
}
