\usepackage{titlesec}

% -----------------------------------------------------------------------------
% 1. MARCADORES DISCRETOS DE SEÇÃO E SUBSEÇÃO (DESIGN EDITORIAL CINEDOOM)
% -----------------------------------------------------------------------------
% Barra vertical em BrandCrimson para Seções Principais (Nível 1)
\newcommand{\editorialBarIcon}{%
  \textcolor{BrandCrimson}{\rule[-0.2ex]{2.2pt}{1.15em}}%
}

% Triângulo apontador em BrandCrimson (#E94560) para Subseções Não Numeradas (Nível 3)
\newcommand{\editorialTriangleIcon}{%
  \raisebox{0.15ex}{\tikz[baseline=-0.35ex]{\fill[BrandCrimson, rounded corners=0.5pt] (0,-0.62ex) -- (1.20ex,0) -- (0,0.62ex) -- cycle;}}%
}

% -----------------------------------------------------------------------------
% 2. NUMERAÇÃO DE SEÇÕES E SUMÁRIO
% -----------------------------------------------------------------------------
% Os títulos já trazem sua própria numeração explícita herdada do Markdown (ex: "1.1 Nome..."),
% portanto secnumdepth é definido como 0 para não duplicar números automáticos ("1.0.1 1.1 ...").
\newcounter{unitcounter}
\setcounter{unitcounter}{0}
\setcounter{secnumdepth}{0}
\setcounter{tocdepth}{3}

% -----------------------------------------------------------------------------
% 3. FORMATAÇÃO DOS TÍTULOS
% -----------------------------------------------------------------------------
% Nível 1: Barra vertical fina em Crimson + Título Midnight Navy #1A1A2E
\titleformat{\section}
  {\normalfont\fontsize{13pt}{16pt}\fontExtraBold\color{BrandNavy}}
  {}
  {0pt}
  {\editorialBarIcon\hspace{7pt}}

% Nível 2: Título Deep Sapphire #0F3460
\titleformat{\subsection}
  {\normalfont\fontsize{11.5pt}{15pt}\fontSemiBold\color{BrandSecondary}}
  {}
  {0pt}
  {}

% Nível 3: Subseção com triângulo apontador BrandCrimson e tipografia BrandNavy
\titleformat{\subsubsection}
  {\normalfont\fontsize{10.5pt}{14pt}\fontSemiBold\color{BrandNavy}}
  {}
  {0pt}
  {\editorialTriangleIcon\hspace{6pt}}

\titleformat{name=\subsubsection,numberless}
  {\normalfont\fontsize{10.5pt}{14pt}\fontSemiBold\color{BrandNavy}}
  {}
  {0pt}
  {\editorialTriangleIcon\hspace{6pt}}

\titleformat{\paragraph}[block]
  {\normalfont\fontsize{10pt}{13.5pt}\fontExtraBold\color{BrandCrimsonMuted}}
  {}
  {0pt}
  {}

\titlespacing*{\section}{0pt}{18pt plus 3pt minus 2pt}{7pt plus 1pt}
\titlespacing*{\subsection}{0pt}{14pt plus 2pt minus 1pt}{5pt plus 1pt}
\titlespacing*{\subsubsection}{0pt}{10pt plus 2pt minus 1pt}{4pt}
\titlespacing*{\paragraph}{0pt}{8pt plus 1pt}{2pt}

% Título de Abertura da Unidade ou Seção Pré/Pós-Textual
\newcommand{\tituloSecaoPrincipal}[1]{%
  \vspace*{6pt}%
  \noindent
  {\fontsize{16pt}{20pt}\fontExtraBold\color{BrandNavy}#1}\par\nointerlineskip
  \vspace{5pt}%
  \noindent\textcolor{BrandCrimson}{\rule{3.5cm}{1.2pt}}\par
  \vspace{12pt}%
}

% Padronização dos Títulos Pré-Textuais nativos (Figuras, Tabelas e Sumário)
\makeatletter
\renewcommand{\listoffigures}{%
  \tituloSecaoPrincipal{\listfigurename}%
  \@starttoc{lof}%
}
\renewcommand{\listoftables}{%
  \tituloSecaoPrincipal{\listtablename}%
  \@starttoc{lot}%
}
\renewcommand{\tableofcontents}{%
  \tituloSecaoPrincipal{\contentsname}%
  \@starttoc{toc}%
}
\makeatother
