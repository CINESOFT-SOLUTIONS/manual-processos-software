% 2. ABERTURA DE CAPÍTULO (ESTRUTURA ORIGINAL COM LOGOS OFICIAIS E TIPOGRAFIA PRECISA)

% - Pontos sutis (halftone) no canto superior direito
% - Cabeçalho com as LOGOS OFICIAIS CineSoft e CineDoom em vetor branco
% - Fita dupla curvada e Poliedro 3D wireframe estilizado
% - Identificador do Capítulo e Título em tipografia geométrica precisa (sem aglutinação)
% - Malha de ondas neon suaves no rodapé

\newcommand{\aberturaunidade}[2]{%
  \clearpage
  \thispagestyle{empty}%
  \global\showcontentbgfalse
  \begin{tikzpicture}[remember picture, overlay]
    % 1. Fundo escuro total (Full-bleed)
    \fill[RefDarkBg] (current page.south west) rectangle (current page.north east);

    % 2. Pontos sutis (halftone) no canto superior direito da página escura
    \foreach \r in {0,...,7} {
      \foreach \c in {0,...,12} {
        \pgfmathsetmacro{\xx}{0.6 + \c*0.42}
        \pgfmathsetmacro{\yy}{0.5 + \r*0.32 + 0.18*sin(\c*25)}
        \pgfmathsetmacro{\op}{max(0.04, 0.28 - \r*0.03 - \c*0.012)}
        \fill[RefNeonPurple!60!white, opacity=\op]
          ($(current page.north east) + (-\xx cm, -\yy cm)$) circle (0.042cm);
      }
    }

    % 3. Cabeçalho Institucional no topo com Logos Oficiais CineSoft & CineDoom
    \begin{scope}[shift={($(current page.north west) + (2.5cm, -2.4cm)$)}]
      % Logo CineSoft horizontal branca
      \node[inner sep=0pt, anchor=west] at (0, 0) {
        \includegraphics[height=0.95cm]{logos/cinesoft-logo-horizontal-white.png}
      };

      % Barra vertical divisória branca
      \draw[white!40, line width=0.8pt] (5.6, 0.45) -- (5.6, -0.45);

      % Logo CineDoom horizontal branca ao lado
      \node[inner sep=0pt, anchor=west] at (6.0, 0) {
        \includegraphics[height=0.95cm]{logos/cinedoom-logo-horizontal-white.png}
      };

      % Barra vertical divisória sutil
      \draw[white!25, line width=0.6pt] (11.5, 0.45) -- (11.5, -0.45);

      % Rótulo técnico institucional à direita
      \node[anchor=west, text=white!85, font=\fontsize{8pt}{10.5pt}\fontRegular, align=left]
        at (11.8, 0) {%
          {\fontSemiBold\color{white}Manual de Processos}\\[1pt]
          {\color{RefNeonCyan}ISO/IEC/IEEE 12207}%
        };
    \end{scope}

    % 4. Fita curva dupla e Poliedro 3D wireframe na zona central
    \begin{scope}[shift={($(current page.west) + (2.6cm, 2.5cm)$)}]
      \filldraw[fill=RefDarkBg, draw=white!85, line width=0.8pt]
        (0.22,-0.32) .. controls (0.72,-0.05) and (1.22,-0.55) .. (1.82,-0.20)
        -- (1.62,0.30) .. controls (1.05,-0.05) and (0.55,0.42) .. (0.04,0.16) -- cycle;
      \filldraw[fill=RefPurple!90, draw=RefLavender, line width=0.8pt]
        (0,0.08) .. controls (0.52,0.38) and (1.02,-0.15) .. (1.62,0.20)
        -- (1.42,0.72) .. controls (0.88,0.38) and (0.38,0.85) .. (-0.18,0.56) -- cycle;
    \end{scope}

    % Poliedro 3D wireframe à direita do título
    \begin{scope}[shift={($(current page.east) + (-4.2cm, 0.3cm)$)}, scale=1.55]
      \begin{scope}[shift={(-1.1, 1.35)}, scale=0.45]
        \filldraw[fill=RefLavender, draw=white, line width=0.5pt]
          (0,0.8) -- (0.7,0.4) -- (0,0) -- (-0.7,0.4) -- cycle;
        \filldraw[fill=RefPurple, draw=white, line width=0.5pt]
          (-0.7,0.4) -- (0,0) -- (0,-0.8) -- (-0.7,-0.4) -- cycle;
        \filldraw[fill=RefPurpleDark, draw=white, line width=0.5pt]
          (0,0) -- (0.7,0.4) -- (0.7,-0.4) -- (0,-0.8) -- cycle;
      \end{scope}
      \coordinate (P1) at (0, 1.45);
      \coordinate (P2) at (1.25, 0.75);
      \coordinate (P3) at (1.35, -0.55);
      \coordinate (P4) at (0.15, -1.40);
      \coordinate (P5) at (-1.15, -0.70);
      \coordinate (P6) at (-1.20, 0.65);
      \coordinate (C1) at (0.20, 0.35);
      \coordinate (C2) at (0.55, -0.45);
      \coordinate (C3) at (-0.45, -0.20);
      \draw[white!28, line width=0.65pt]
        (P1)--(P2)--(P3)--(P4)--(P5)--(P6)--cycle
        (P1)--(C1)--(P2) (P2)--(C2)--(P3) (P3)--(C2)--(P4)
        (P4)--(C3)--(P5) (P5)--(C3)--(P6) (P6)--(C1)--(P1)
        (C1)--(C2)--(C3)--cycle;
      \foreach \pt in {P1,P2,P3,P4,P5,P6,C1,C2,C3} {
        \fill[RefLavender!60, opacity=0.55] (\pt) circle (0.04cm);
      }
    \end{scope}

    % 5. Tipografia Central de Abertura (com espaçamento impecável entre palavras)
    \node[
      anchor=north west,
      inner sep=0pt
    ] at ($(current page.west) + (2.5cm, 1.8cm)$) {%
      \parbox{15.0cm}{%
        {\fontsize{24pt}{29pt}\fontLight\color{RefLavender} #1\par}\vspace{14pt}
        {\fontsize{28pt}{34pt}\fontExtraBold\color{white} #2\par}%
      }%
    };

    % 6. Malha de Ondas Wireframe Neon (Azul / Ciano / Roxo) no rodapé
    \begin{scope}[shift={(current page.south west)}]
      \foreach \i in {1,...,22} {
        \pgfmathsetmacro{\phase}{\i*6.5}
        \pgfmathsetmacro{\ampA}{1.8 + \i*0.14}
        \pgfmathsetmacro{\ampB}{0.9 + \i*0.11}
        \pgfmathsetmacro{\ybase}{1.2 + \i*0.24}
        \pgfmathsetmacro{\mix}{min(100, \i*4.5)}
        \draw[RefNeonBlue!\mix!RefNeonPurple, line width=0.55pt, opacity=0.48]
          plot[domain=0:21, samples=60, smooth]
          (\x, {\ybase + 1.35*sin(\x*16 + \phase) + 0.65*cos(\x*24 - \phase*0.7)});
      }
      \foreach \xgrid in {1.2, 2.8, 4.5, 6.2, 8.0, 9.8, 11.6, 13.5, 15.4, 17.2, 19.2} {
        \pgfmathsetmacro{\ytop}{3.8 + 1.35*sin(\xgrid*16 + 110) + 0.65*cos(\xgrid*24 - 75)}
        \pgfmathsetmacro{\ymid}{2.6 + 1.35*sin(\xgrid*16 + 65) + 0.65*cos(\xgrid*24 - 45)}
        \pgfmathsetmacro{\ybot}{1.5 + 1.35*sin(\xgrid*16 + 20) + 0.65*cos(\xgrid*24 - 15)}
        \draw[RefNeonCyan!70!white, line width=0.4pt, opacity=0.35]
          (\xgrid, \ybot) -- (\xgrid+0.4, \ymid) -- (\xgrid-0.2, \ytop);
        \fill[RefNeonCyan, opacity=0.75] (\xgrid, \ybot) circle (0.06cm);
        \fill[white, opacity=0.90] (\xgrid+0.4, \ymid) circle (0.05cm);
        \fill[RefNeonPurple!60!white, opacity=0.80] (\xgrid-0.2, \ytop) circle (0.055cm);
      }
    \end{scope}
  \end{tikzpicture}%
  \mbox{}%
  \clearpage
  \global\showcontentbgtrue
  \stepcounter{unitcounter}%
