% -----------------------------------------------------------------------------
% 2. ABERTURA DE CAPÍTULO (FUNDO EM PDF + TEXTO DINÂMICO CENTRALIZADO)
% Formato: "Capítulo X \n Nome do Capítulo"
% - "Capítulo X" menor e em cor suave e elegante (white!80 / RefLavender)
% - "Nome do Capítulo" em destaque branco (\fontExtraBold)
% -----------------------------------------------------------------------------
\newcommand{\aberturaunidade}[2]{%
  \clearpage
  \thispagestyle{empty}%
  \global\showcontentbgfalse
  \begin{tikzpicture}[remember picture, overlay]
    % 1. Fundo artístico da divisória de capítulo (Full-bleed A4)
    \node[inner sep=0pt, outer sep=0pt] at (current page.center) {%
      \includegraphics[width=\paperwidth, height=\paperheight]{capas/abertura-capitulo.pdf}%
    };

    % 2. Bloco Central de Título Dinâmico (Perfeitamente centralizado ao meio do PDF)
    \node[
      anchor=center,
      inner sep=0pt,
      align=center
    ] at ($(current page.center) + (0, -0.2cm)$) {%
      \parbox{16.5cm}{\centering
        {\fontsize{17pt}{19pt}\fontSemiBold\color{white!80}\MakeUppercase{#1}\par}\vspace{-4pt}
        {\fontsize{40pt}{46pt}\fontExtraBold\color{white} #2\par}%
      }%
    };
  \end{tikzpicture}%
  \mbox{}%
  \clearpage
  \global\showcontentbgtrue
  \stepcounter{unitcounter}%
  \setcounter{section}{0}%
  \addUnitToTOC{#1}{#2}%
  \tituloSecaoPrincipal{#1 — #2}%
}

% Alias semântico para Capítulos do Manual
\newcommand{\aberturacapitulo}[2]{\aberturaunidade{#1}{#2}}
